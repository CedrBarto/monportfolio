{{- /* Image Markdown : cadre du site, aperçu réduit qu'on peut agrandir, légende et lien vers la version HD si un fichier "-hd" existe */ -}} {{- $src := .Destination -}} {{- $remote := or (strings.HasPrefix $src "http://") (strings.HasPrefix $src "https://") -}} {{- $url := $src -}} {{- $hd := "" -}} {{- $w := 0 -}}{{- $h := 0 -}} {{- if not $remote -}} {{- $src = strings.TrimPrefix "/" $src -}} {{- $url = $src | relURL -}} {{- $file := path.Join "static" $src -}} {{- if fileExists $file -}} {{- with imageConfig $file -}}{{- $w = .Width -}}{{- $h = .Height -}}{{- end -}} {{- end -}} {{- $hdSrc := replaceRE `\.(\w+)$` "-hd.$1" $src -}} {{- if fileExists (path.Join "static" $hdSrc) -}}{{- $hd = $hdSrc | relURL -}}{{- end -}} {{- end -}}
